cmake_minimum_required(VERSION 3.22)
project(kadunce_unload_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(KF6 REQUIRED COMPONENTS CoreAddons Config)
find_package(KWin REQUIRED COMPONENTS kwineffects)
find_package(Wayland REQUIRED COMPONENTS Server)
find_package(KDecoration3 REQUIRED)
find_package(XCB REQUIRED COMPONENTS XCB)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
add_library(kadunce_unload_probe MODULE probe.cpp ../../native/src/WorkspaceInputRouter.cpp ../../native/src/CardLineLayout.cpp ../../native/src/DesktopStageController.cpp ../../native/src/BentoLayout.cpp)
target_sources(kadunce_unload_probe PRIVATE ../../native/src/CardStageController.cpp ../../native/src/CardLineModel.cpp)
set_target_properties(kadunce_unload_probe PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/kwin/effects/plugins")
# Deliberately no install rules: this input injector is only for a private test bus.
target_include_directories(kadunce_unload_probe PRIVATE ../../native/src)
target_link_libraries(kadunce_unload_probe PRIVATE KWin::kwin Qt6::DBus KF6::ConfigCore Wayland::Server KDecoration3::KDecoration XCB::XCB)
add_executable(unload-client client.cpp)
target_link_libraries(unload-client PRIVATE Qt6::Widgets Qt6::DBus XCB::XCB)
